完整优化方案分层（文档答案要点）

  • 数据侧：LTTB（三角形面积最大，保形状/峰谷）、min-max（每桶保留极值，告警峰值不丢）；缩放时按视窗重新降采样并按层级缓存。
  • 渲染侧：单条海量曲线用 WebGL 批量绘制（ECharts GL / regl / 自建 shader），顶点一次性提交 GPU；纯 DOM/SVG 在百万量级不可行。
  • 列表层：虚拟滚动，不可见图表销毁实例或留占位（上方即最小实现）。
  • 计算侧：降采样/统计聚合移入 Web Worker，大数组用 Transferable(ArrayBuffer) 传输。
  • 调度侧：首屏只画可视区间，其余 requestIdleCallback / rAF 分片初始化，单帧超 8ms 预算就让出主线程。
  • 交互侧：tooltip 查值用二分/预建索引；降采样后展示「已抽稀」提示保证数据诚实。

LTTB 核心实现（本页实际使用）

// Largest-Triangle-Three-Buckets：每桶选与「前一选中点 + 下一桶均值」构成三角形面积最大的点
function lttb(data, threshold) {
  const n = data.length;
  if (threshold >= n || threshold < 3) return data.slice();
  const out = [data[0]];
  const bucket = (n - 2) / (threshold - 2);
  let a = 0;
  for (let i = 0; i < threshold - 2; i++) {
    let sx = 0, sy = 0; const s0 = Math.floor((i + 1) * bucket) + 1;
    const s1 = Math.min(Math.floor((i + 2) * bucket) + 1, n - 1);
    for (let j = s0; j < s1; j++) { sx += data[j][0]; sy += data[j][1]; }
    const cnt = s1 - s0, ax = sx / cnt, ay = sy / cnt;       // 下一桶均值
    const r0 = Math.floor(i * bucket) + 1, r1 = Math.floor((i + 1) * bucket) + 1;
    let best = -1, bestArea = -1;
    for (let j = r0; j < r1; j++) {                           // 当前桶内找最大面积
      const area = Math.abs((data[a][0] - ax) * (data[j][1] - data[a][1])
                          - (data[a][0] - data[j][0]) * (ay - data[a][1])) * 0.5;
      if (area > bestArea) { bestArea = area; best = j; }
    }
    out.push(data[best]); a = best;
  }
  out.push(data[n - 1]);
  return out;
}
